{{/* estado.html — calcula el estado del experimento y lo devuelve como dict. Uso: {{ $e := partial "estado.html" . }} Fuentes: data/estado.yaml (encargos, cobrado, tramos) y params.inicio/fin. */}} {{- $d := hugo.Data.estado -}} {{- $inicio := time.AsTime site.Params.inicio -}} {{- $fin := time.AsTime site.Params.fin -}} {{- $total := 90 -}} {{- $dia := add 1 (div (sub now.Unix $inicio.Unix) 86400) -}} {{- if lt $dia 1 }}{{ $dia = 1 }}{{ end -}} {{- if gt $dia $total }}{{ $dia = $total }}{{ end -}} {{- $tramo := index $d.tramos 0 -}} {{- $siguiente := false -}} {{- range $d.tramos -}} {{- $desde := time.AsTime .desde -}} {{- $hasta := (time.AsTime .hasta).Add (time.ParseDuration "23h59m") -}} {{- if and (ge now.Unix $desde.Unix) (le now.Unix $hasta.Unix) -}} {{- $tramo = . -}} {{- end -}} {{- end -}} {{- range $d.tramos -}} {{- if and (not $siguiente) (gt (time.AsTime .desde).Unix now.Unix) -}} {{- $siguiente = . -}} {{- end -}} {{- end -}} {{- return dict "dia" $dia "total" $total "tramo" $tramo "tramoSiguiente" $siguiente "tramos" $d.tramos "encargos" ($d.encargos | default 0) "cobrado" ($d.cobrado | default 0) "actualizado" $d.actualizado "inicio" $inicio "fin" $fin "postCero" (time.AsTime site.Params.post_cero) "empezado" (ge now.Unix $inicio.Unix) "terminado" (gt now.Unix ($fin.Add (time.ParseDuration "23h59m")).Unix) -}}
