{{/* estado.html — calcula el estado del experimento y lo devuelve como dict. Uso: {{ $e := partial "estado.html" . }} Fuentes: data/estado.yaml (encargos, cobrado, tramos) y params.inicio/fin. */}} {{- $d := hugo.Data.estado -}} {{- $inicio := time.AsTime site.Params.inicio -}} {{- $fin := time.AsTime site.Params.fin -}} {{- $total := 90 -}} {{/* El día se cuenta por fechas de calendario en la zona del sitio (hugo.toml → timeZone), no por segundos desde params.inicio: como ese instante es las 09:00, el contador iba un día atrasado hasta esa hora, que incluye la build diaria de las 02:23 y el volcado de las 06:30. Así tampoco depende de la zona en la que corra Hugo. */}} {{- $zona := (time.AsTime "2000-01-01").Location -}} {{- $hoy := (now.In $zona).Format "2006-01-02" -}} {{- $dia0 := ($inicio.In $zona).Format "2006-01-02" -}} {{- $dia := add 1 (div (sub (time.AsTime $hoy).Unix (time.AsTime $dia0).Unix) 86400) -}} {{- if lt $dia 1 }}{{ $dia = 1 }}{{ end -}} {{- if gt $dia $total }}{{ $dia = $total }}{{ end -}} {{- $tramo := index $d.tramos 0 -}} {{- $siguiente := false -}} {{- range $d.tramos -}} {{- $desde := time.AsTime .desde -}} {{- $hasta := (time.AsTime .hasta).Add (time.ParseDuration "23h59m") -}} {{- if and (ge now.Unix $desde.Unix) (le now.Unix $hasta.Unix) -}} {{- $tramo = . -}} {{- end -}} {{- end -}} {{- range $d.tramos -}} {{- if and (not $siguiente) (gt (time.AsTime .desde).Unix now.Unix) -}} {{- $siguiente = . -}} {{- end -}} {{- end -}} {{- return dict "dia" $dia "total" $total "tramo" $tramo "tramoSiguiente" $siguiente "tramos" $d.tramos "encargos" ($d.encargos | default 0) "cobrado" ($d.cobrado | default 0) "ganado" ($d.ganado | default 0) "actualizado" $d.actualizado "inicio" $inicio "fin" $fin "postCero" (time.AsTime site.Params.post_cero) "empezado" (ge now.Unix $inicio.Unix) "terminado" (gt now.Unix ($fin.Add (time.ParseDuration "23h59m")).Unix) -}}
